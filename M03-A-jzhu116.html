def export_and_display_figure(fig, name, width=1200, height=700, scale=1):
    """Save interactive HTML and display an interactive or static figure."""
    html_path = OUTPUT_DIR / f"{name}.html"
    png_path = OUTPUT_DIR / f"{name}.png"

    fig.write_html(
        str(html_path),
        full_html=True,
        include_plotlyjs="cdn",
    )

    if not STATIC_EXPORT:
        display(fig)
        return

    try:
        fig.write_image(
            str(png_path),
            width=width,
            height=height,
            scale=scale,
        )
        display(Image(filename=str(png_path)))
    except Exception as exc:
        print(f"Static export failed for {name}: {exc}")
        print("Displaying the interactive Plotly figure instead.")
        display(fig)


def stratified_plot_sample(
    spark_df,
    group_column,
    selected_columns,
    maximum_rows_per_group,
    seed=42,
):
    """Collect a bounded stratified sample while leaving full data in Spark."""
    group_count_rows = spark_df.groupBy(group_column).count().collect()
    sampling_fractions = {}
    target_rows = 0

    for row in group_count_rows:
        group_value = row[group_column]
        group_count = int(row["count"])
        if group_value is None or group_count <= 0:
            continue

        target_count = min(maximum_rows_per_group, group_count)
        sampling_fractions[group_value] = target_count / group_count
        target_rows += target_count

    if not sampling_fractions:
        return pd.DataFrame(columns=selected_columns)

    sample_pdf = (
        spark_df
        .sampleBy(group_column, fractions=sampling_fractions, seed=seed)
        .select(*selected_columns)
        .toPandas()
    )

    print(
        f"Plot sample collected: {len(sample_pdf):,} rows "
        f"(target approximately {target_rows:,})."
    )
    return sample_pdf

3.2 1.2 Cast Salary and Experience Columns

numeric_columns = [

clean_df = df

    numeric_expression = (
        f"try_cast(regexp_replace(trim(`{column_name}`), '[,$]', '') as double)"
    )
    clean_df = clean_df.withColumn(
        column_name,
        F.expr(numeric_expression),

The salary and experience fields are converted from strings to numeric values so they can be summarized and visualized correctly. try_cast safely converts malformed or structured salary strings to missing values instead of stopping the Spark job.

3.3 1.3 Compute Medians and Impute Missing Salaries

median_row = clean_df.agg(
    F.percentile_approx("SALARY_FROM", 0.5).alias("median_from"),
    F.percentile_approx("SALARY_TO", 0.5).alias("median_to"),
    F.percentile_approx("SALARY", 0.5).alias("median_salary"),
).first()

median_from = median_row["median_from"]
median_to = median_row["median_to"]
median_salary = median_row["median_salary"]

print("Medians:", median_from, median_to, median_salary)

if median_from is None or median_to is None:
    raise ValueError("Salary-bound medians could not be calculated.")

clean_df = (
    clean_df
    .withColumn(
        "SALARY_FROM",
        F.when(
            F.col("SALARY_FROM").isNull() | (F.col("SALARY_FROM") <= 0),
            F.lit(float(median_from)),
        ).otherwise(F.col("SALARY_FROM")),
    )
    .withColumn(
        "SALARY_TO",
        F.when(
            F.col("SALARY_TO").isNull() | (F.col("SALARY_TO") <= 0),
            F.lit(float(median_to)),
        ).otherwise(F.col("SALARY_TO")),
    )
    .withColumn(
        "Average_Salary",
        (F.col("SALARY_FROM") + F.col("SALARY_TO")) / 2.0,
    )
    .withColumn(
        "SALARY",
        F.coalesce(F.col("SALARY"), F.col("Average_Salary")),
    )
)

def plot_remote_histogram(group_name):
    group_pdf = remote_plot_pdf.loc[
        remote_plot_pdf["Remote_Group"] == group_name
    ].copy()

        return

    salary_values = pd.to_numeric(
        group_pdf["Average_Salary"],
        errors="coerce",
    ).to_numpy(dtype=float)
    salary_values = salary_values[
        np.isfinite(salary_values) & (salary_values > 0)
    ]

    if salary_values.size == 0:
        display(Markdown(f"**Key insights.** No valid salary values were available for {group_name}."))
        return

    histogram_counts, histogram_edges = np.histogram(
        salary_values,
        bins=remote_histogram_bin_edges,
    )
    nonempty_bins = histogram_counts > 0
    histogram_left_edges = histogram_edges[:-1][nonempty_bins]
    histogram_right_edges = histogram_edges[1:][nonempty_bins]
    histogram_widths = (
        histogram_right_edges - histogram_left_edges
    )

    if remote_log_scale:
        histogram_centers = np.sqrt(
            histogram_left_edges * histogram_right_edges
        )
    else:
        histogram_centers = (
            histogram_left_edges + histogram_right_edges
        ) / 2.0

    histogram_hover_data = np.column_stack(
        [histogram_left_edges, histogram_right_edges]
    )

    remote_histogram = go.Figure(
        go.Bar(
            x=histogram_centers,
            y=histogram_counts[nonempty_bins],
            width=histogram_widths,

                "color": remote_colors[group_name],
                "line": {"color": "#FFFFFF", "width": 0.7},
